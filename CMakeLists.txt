cmake_minimum_required(VERSION 3.21)
project(myumiq_feasibility LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This spike targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENVR_SDK "${CMAKE_SOURCE_DIR}/.cache/openvr" CACHE PATH "Official OpenVR SDK checkout")
if(NOT EXISTS "${OPENVR_SDK}/headers/openvr_driver.h")
  message(FATAL_ERROR "Set OPENVR_SDK to an OpenVR SDK checkout; see docs/feasibility.md")
endif()
add_library(driver_myumiq SHARED native/driver.cpp)
target_include_directories(driver_myumiq PRIVATE "${OPENVR_SDK}/headers")
set_target_properties(driver_myumiq PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/myumiq/bin/win64" PREFIX "")
file(COPY driver/ DESTINATION "${CMAKE_BINARY_DIR}/myumiq")
add_executable(pose_probe native/pose_probe.cpp)
target_include_directories(pose_probe PRIVATE "${OPENVR_SDK}/headers")
target_link_libraries(pose_probe PRIVATE "${OPENVR_SDK}/lib/win64/openvr_api.lib")
add_custom_command(TARGET pose_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${OPENVR_SDK}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:pose_probe>")
add_executable(frame_probe native/frame_probe.cpp)
target_include_directories(frame_probe PRIVATE "${OPENVR_SDK}/headers")
target_link_libraries(frame_probe PRIVATE "${OPENVR_SDK}/lib/win64/openvr_api.lib")
add_custom_command(TARGET frame_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${OPENVR_SDK}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:frame_probe>")
# SteamVR requires a flat bin/win64 directory even with multi-config generators.
add_custom_command(TARGET driver_myumiq POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "$<TARGET_FILE:driver_myumiq>" "${CMAKE_BINARY_DIR}/myumiq/bin/win64/driver_myumiq.dll")
enable_testing()
add_executable(native_tests native/tests.cpp)
target_include_directories(native_tests PRIVATE "${OPENVR_SDK}/headers")
add_test(NAME native_contracts COMMAND native_tests)
find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
add_test(NAME binding_contracts COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -File "${CMAKE_SOURCE_DIR}/scripts/check-bindings.ps1")
